export type SessionName = 'ASIA' | 'LONDON' | 'NEW_YORK' | 'OVERLAP_LD_NY' | 'CLOSED';

export interface SessionWindow {
  readonly name: SessionName;
  readonly fromUtcHour: number;
  readonly toUtcHour: number;
}

export interface SessionInfo {
  readonly session: SessionName;
  readonly is_market_open: boolean;
  readonly is_weekend: boolean;
  readonly is_high_liquidity: boolean;
  readonly utc_hour: number;
  readonly utc_day_of_week: number;
}

/** آ§7:Session Calendar â€” ظٹط­ط¯ط¯ ظ…ط§ ط¥ط°ط§ ظƒط§ظ† ط§ظ„ظˆظ‚طھ ط¶ظ…ظ† ط³ظˆظ‚ ظ…ظپطھظˆط­ */
export function classifySession(at: Timestamp): SessionInfo {
  const d = new Date(at);
  const hour = d.getUTCHours();
  const dow = d.getUTCDay(); // 0=Sunday, 6=Saturday

  const is_weekend = dow === 6 || (dow === 0 && hour < 22);

  let session: SessionName = 'CLOSED';
  if (!is_weekend) {
    // Overlap London+NY: 13-17
    if (hour >= 13 && hour < 17) session = 'OVERLAP_LD_NY';
    else if (hour >= 8 && hour < 17) session = 'LONDON';
    else if (hour >= 13 && hour < 22) session = 'NEW_YORK';
    else if (hour >= 0 && hour < 9) session = 'ASIA';
    else session = 'CLOSED';
  }

  const is_market_open = !is_weekend;
  const is_high_liquidity = session === 'OVERLAP_LD_NY' || session === 'LONDON' || session === 'NEW_YORK';

  return {
    session,
    is_market_open,
    is_weekend,
    is_high_liquidity,
    utc_hour: hour,
    utc_day_of_week: dow,
  };
}

/** آ§6.2: ظٹظ…ظٹط² gap ط¹ط·ظ„ط© ط¹ظ† gap ط®ط·ط£ طھط؛ط°ظٹط© */
export type GapType = 'EXPECTED_MARKET_CLOSED' | 'UNEXPECTED_FEED_ERROR';

export interface GapAnalysis {
  readonly from: Timestamp;
  readonly to: Timestamp;
  readonly duration_ms: number;
  readonly expected_candles: number;
  readonly gap_type: GapType;
  readonly reason: string;
}

export function classifyGap(
  fromUtc: Timestamp,
  toUtc: Timestamp,
  timeframeMs: number,
): GapAnalysis {
  const from = new Date(fromUtc);
  const to = new Date(toUtc);
  const duration_ms = to.getTime() - from.getTime();
  const expected_candles = Math.max(0, Math.round(duration_ms / timeframeMs) - 1);

  // آ§6.2: ط§ظ„ظپط¬ظˆط© ط¹ط·ظ„ط© ظ†ظ‡ط§ظٹط© ط§ظ„ط£ط³ط¨ظˆط¹ = ظ…طھظˆظ‚ط¹ط©
  let touchesWeekend = false;
  let cursor = new Date(from);
  let guard = 0;
  while (cursor < to && guard < 20_000) {
    const dow = cursor.getUTCDay();
    if (dow === 6 || dow === 0) { touchesWeekend = true; break; }
    cursor = new Date(cursor.getTime() + timeframeMs);
    guard++;
  }

  if (touchesWeekend && duration_ms <= 3 * 24 * 60 * 60 * 1000) {
    return {
      from: fromUtc, to: toUtc, duration_ms, expected_candles,
      gap_type: 'EXPECTED_MARKET_CLOSED',
      reason: 'weekend close',
    };
  }

  if (expected_candles > 60) {
    return {
      from: fromUtc, to: toUtc, duration_ms, expected_candles,
      gap_type: 'UNEXPECTED_FEED_ERROR',
      reason: 'gap > 60 candles without weekend',
    };
  }

  return {
    from: fromUtc, to: toUtc, duration_ms, expected_candles,
    gap_type: 'UNEXPECTED_FEED_ERROR',
    reason: 'intraday gap',
  };
}
import { describe, it, expect } from 'vitest';
import { classifySession, classifyGap } from '../calendar/session-calendar';

describe('Session Calendar (آ§7)', () => {
  // Monday 2025-01-06 00:00 UTC = session ASIA
  const mon00 = new Date(Date.UTC(2025, 0, 6, 0, 0, 0)).toISOString();
  const mon10 = new Date(Date.UTC(2025, 0, 6, 10, 0, 0)).toISOString();
  const mon15 = new Date(Date.UTC(2025, 0, 6, 15, 0, 0)).toISOString();
  const sat12 = new Date(Date.UTC(2025, 0, 4, 12, 0, 0)).toISOString();

  it('Monday 00:00 = ASIA', () => {
    const s = classifySession(mon00);
    expect(s.session).toBe('ASIA');
    expect(s.is_market_open).toBe(true);
    expect(s.is_weekend).toBe(false);
  });

  it('Monday 10:00 = LONDON', () => {
    expect(classifySession(mon10).session).toBe('LONDON');
    expect(classifySession(mon10).is_high_liquidity).toBe(true);
  });

  it('Monday 15:00 = OVERLAP (London+NY)', () => {
    expect(classifySession(mon15).session).toBe('OVERLAP_LD_NY');
    expect(classifySession(mon15).is_high_liquidity).toBe(true);
  });

  it('Saturday = market closed', () => {
    const s = classifySession(sat12);
    expect(s.is_weekend).toBe(true);
    expect(s.is_market_open).toBe(false);
  });
});

describe('Gap Classifier (آ§6.2)', () => {
  const mon00 = new Date(Date.UTC(2025, 0, 6, 0, 0, 0)).toISOString();
  const mon01 = new Date(Date.UTC(2025, 0, 6, 1, 0, 0)).toISOString();
  const fri22 = new Date(Date.UTC(2025, 0, 3, 22, 0, 0)).toISOString();
  const mon00_next = new Date(Date.UTC(2025, 0, 6, 0, 0, 0)).toISOString();

  it('small intraday gap = feed error', () => {
    const g = classifyGap(mon00, mon01, 60_000);
    expect(g.gap_type).toBe('UNEXPECTED_FEED_ERROR');
    expect(g.expected_candles).toBe(59);
  });

  it('weekend gap = expected closed', () => {
    const g = classifyGap(fri22, mon00_next, 60_000);
    expect(g.gap_type).toBe('EXPECTED_MARKET_CLOSED');
    expect(g.reason).toContain('weekend');
  });
});
